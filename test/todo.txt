import { useRef, useEffect, useState, createContext, useContext } from "react";
import { io } from "socket.io-client";

const SocketContext = createContext();

export const SocketProvider = ({ children }) => {
  const socketRef = useRef(null);
  const [updates, setUpdates] = useState({}); // { centerId: [areas...] }

  useEffect(() => {
    const evacuationCenterId = localStorage.getItem("evacuationCenterId");
    console.log("📝 Evacuation Center ID from localStorage:", evacuationCenterId);
    if (!evacuationCenterId) return;

    if (!socketRef.current) {
      console.log("⚡ Initializing socket connection...");

      const socket = io("http://localhost:3000", {
        transports: ["websocket", "polling"],
        reconnection: true,
        reconnectionAttempts: 10,
        reconnectionDelay: 1000,
      });

      socket.on("connect", () => {
        console.log("✅ Socket connected:", socket.id);
        socket.emit("joinRoom", `area-${evacuationCenterId}`);
      });

      socket.on("connect_error", (err) => {
        console.error("❌ Socket connection error:", err.message);
      });

      socket.on("disconnect", (reason) => {
        console.log("⚠️ Socket disconnected:", reason);
      });

      socket.on("centerAreaListUpdate", (data) => {
        console.log("📡 Received center area update:", data);
        const centerId = data.evacuation_center_id;

        setUpdates((prev) => {
          const centerUpdates = prev[centerId] ? [...prev[centerId]] : [];

          if (data.deleted) {
            return {
              ...prev,
              [centerId]: centerUpdates.filter((area) => area._id !== data._id),
            };
          } else {
            const index = centerUpdates.findIndex((area) => area._id === data._id);
            if (index > -1) centerUpdates[index] = data;
            else centerUpdates.push(data);

            return { ...prev, [centerId]: centerUpdates };
          }
        });
      });

      socketRef.current = socket;
    }

    return () => {
      if (socketRef.current && socketRef.current.connected) {
        console.log("🧹 Disconnecting socket on unmount");
        socketRef.current.disconnect();
      }
    };

  }, []);

  return (
    <SocketContext.Provider value={{ socket: socketRef.current, updates }}>
      {children}
    </SocketContext.Provider>
  );
};

export const useSocket = () => useContext(SocketContext);

import { useEffect } from 'react';

const VAPID_PUBLIC_KEY = 'BNjOeGzU4I84a4vn9OnPnv1dVYj5XvCskv8cpiVjWwsnFsUJ9x5PXMTWIbTdqKJqVzC9tYaIKapX3NsbyAY6nuo';

function urlBase64ToUint8Array(base64String) {
  const padding = '='.repeat((4 - base64String.length % 4) % 4);
  const base64 = (base64String + padding)
    .replace(/\-/g, '+')
    .replace(/_/g, '/');

  const raw = atob(base64);
  return new Uint8Array([...raw].map(c => c.charCodeAt(0)));
}

export default function useNotificationSetup(user) {
  const { socket } = useSocket();

  useEffect(() => {
    if (!user || !user._id || !user.role || !socket) return;

    // Register user for socket notifications
    socket.emit('register', { userId: user._id, role: user.role });

    socket.on('notification', (data) => {
      toast(`${data.title}: ${data.body}`, {
        position: 'top-right',
        type: 'info',
        autoClose: 5000,
      });
    });

    // Push notifications
    if ('serviceWorker' in navigator && 'PushManager' in window) {
      navigator.serviceWorker.register('/service-worker.js')
        .then(reg => {
          Notification.requestPermission().then(permission => {
            if (permission === 'granted') {
              reg.pushManager.subscribe({
                userVisibleOnly: true,
                applicationServerKey: urlBase64ToUint8Array(VAPID_PUBLIC_KEY)
              }).then(subscription => {
                // Save subscription to backend
                fetch('http://localhost:3000/push-subscription', {
                  method: 'POST',
                  headers: { 'Content-Type': 'application/json' },
                  body: JSON.stringify({ user_id: user._id, role: user.role, subscription })
                }).catch(console.error);
              }).catch(console.error);
            }
          });
        });
    }

    return () => socket.off('notification');
  }, [user, socket]);
}
